     * بيع كاش

    public void postCashSale(BigDecimal amount, String description) {

        journal.post(entry);

    /*
     * إنشاء رأس القيد
     */
    private JournalEntry createEntry(String description) {

        return JournalEntry.builder()
                .entryDate(LocalDateTime.now())
                .description(description)
                .build();
    }

    /*
     * إضافة مدين
     */
    private void addDebit(JournalEntry entry, Account account, BigDecimal amount) {

        JournalEntryLine line = JournalEntryLine.builder()
                .journalEntry(entry)
                .account(account)
                .debit(amount)
                .credit(BigDecimal.ZERO)
                .build();

        entry.getLines().add(line);
    }

    /*
     * إضافة دائن
     */
    private void addCredit(JournalEntry entry, Account account, BigDecimal amount) {

        JournalEntryLine line = JournalEntryLine.builder()
                .journalEntry(entry)
                .account(account)
                .debit(BigDecimal.ZERO)
                .credit(amount)
                .build();

        entry.getLines().add(line);
    }
